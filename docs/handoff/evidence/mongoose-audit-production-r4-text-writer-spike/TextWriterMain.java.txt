package com.example.myapp.spike;

import com.telamin.fluxtion.runtime.audit.LogRecordListener;
import com.telamin.mongoose.MongooseServer;

import java.io.FileReader;
import java.nio.file.Path;

/** REVIEW SPIKE. Boots the bundle's own server-config.yml with the text writer as the server's listener.
 *  -Dtw.order=stopFirst (default): SIGTERM stops the server, THEN writes the marker.
 *  -Dtw.order=markerFirst: writes the marker, THEN stops the server — the wrong order, to see it fail. */
public final class TextWriterMain {
    public static void main(String[] args) throws Exception {
        String cfg = System.getProperty("mongooseServer.config.file", "config/server-config.yml");
        System.setProperty("mongooseServer.config.file", cfg);
        boolean flushEach = !"false".equals(System.getProperty("tw.flushEach"));
        String order = System.getProperty("tw.order", "stopFirst");
        LogRecordListener console = r -> System.out.println("[console] " + r.asCharSequence().length() + " chars");
        TextAuditWriter writer = new TextAuditWriter(Path.of(System.getProperty("tw.dir","audit-text")), console, flushEach);
        MongooseServer server;
        try (FileReader reader = new FileReader(cfg)) {
            server = MongooseServer.bootServer(reader, writer);
        }
        Runtime.getRuntime().addShutdownHook(new Thread(() -> {
            long t0=System.currentTimeMillis();
            try {
                if ("markerFirst".equals(order)) { writer.close("stopping"); server.stop(); }
                else { server.stop(); writer.close("stopping"); }
                System.out.println("records arriving after the marker: " + writer.afterClose() + " stopMillis=" + (System.currentTimeMillis()-t0));
            } catch (Exception e) { e.printStackTrace(); }
        }, "tw-shutdown"));
        System.out.println("TW started, writing " + writer.file() + " order=" + order + " flushEach=" + flushEach);
    }
}
