    static String xmlUnused(boolean include) { return ExtendedDeclarationsTest.xml("DATA").replace("<value>java.lang.String</value>","<value>java.lang.String</value>"+(include?"<value>demo.event.Unused</value>":"")); }
    static Path unused(Path root) { return root.resolve("src/main/java/demo/event/Unused.java"); }
    static String rec(Path root) throws Exception { Path p=root.resolve("fluxtion-authoring.json"); return Files.exists(p)?Files.readString(p):null; }

    Reconciler.Plan firstWithUnused(Path root) throws Exception {
        var first=fixture.plan(root,xmlUnused(true),fixture.record()); assertFalse(first.conflicts()); Reconciler.apply(root,first,null); assertTrue(Files.exists(unused(root))); return first;
    }
    void withdrawUnused(Path root, Reconciler.Plan first) throws Exception {
        var next=fixture.plan(root,xmlUnused(false),first.record()); assertFalse(StarterJson.write(next.report()),next.conflicts()); Reconciler.apply(root,next,rec(root));
    }

    // ---------- 6. transactionality ----------
    @Test public void tx_recordWriteFailureRestoresDeletedTypeAndNewFile() throws Exception {
        Path root=tmp.newFolder().toPath().toRealPath(); var first=firstWithUnused(root);
        String orig=Files.readString(unused(root)), recBefore=rec(root);
        String xml=xmlUnused(false).replace("<value>java.lang.String</value>","<value>java.lang.String</value><value>demo.fresh.Brand</value>");
        var next=fixture.plan(root,xml,first.record()); assertFalse(StarterJson.write(next.report()),next.conflicts());
        assertTrue("plan should delete Unused", next.sources().containsKey(unused(root)) && next.sources().get(unused(root))==null);
        Path fresh=root.resolve("src/main/java/demo/fresh/Brand.java"); assertTrue("plan should create Brand: "+next.sources().keySet(), next.sources().containsKey(fresh));
        // make record write fail: project root not writable (temp file for the record cannot be created)
        root.toFile().setWritable(false);
        try { Reconciler.apply(root,next,recBefore); fail("record write should fail"); } catch(java.io.IOException expected) {}
        finally { root.toFile().setWritable(true); }
        assertEquals("deleted whole type restored", orig, Files.readString(unused(root)));
        assertFalse("new file rolled back", Files.exists(fresh));
        assertEquals("record unchanged", recBefore, rec(root));
        System.out.println("TX leftover dir demo/fresh exists=" + Files.exists(fresh.getParent()));
    }

    // ---------- 2. G12 withdrawal field selection ----------
    String planAndShow(Path root, String xml, String label) throws Exception {
        var p=fixture.plan(root,xml,fixture.record());
        String child=p.sources().getOrDefault(root.resolve("src/main/java/demo/node/Child.java"),"<unchanged>");
        System.out.println("=== "+label+" conflicts="+p.conflicts()+"\n"+(p.conflicts()?diag(p):child));
        if(!p.conflicts()) { Reconciler.apply(root,p,null); }
        return p.conflicts()?null:Files.readString(root.resolve("src/main/java/demo/node/Child.java"));
    }
    static String diag(Reconciler.Plan p) throws Exception { String j=StarterJson.write(p.report().diagnosticReport()); return j.length()>2500?j.substring(0,2500):j; }
    Path childDir(Path root) throws Exception { return Files.createDirectories(root.resolve("src/main/java/demo/node")); }

        Path root=tmp.newFolder().toPath().toRealPath(); Path d=childDir(root);

        Files.writeString(d.resolve("Parent.java"),"package demo.node; public class Parent implements ParentApi {}\n");

    void repeatStable(String orig, String label) throws Exception {
        Path root=tmp.newFolder().toPath().toRealPath(); Path d=childDir(root); Files.writeString(d.resolve("Parent.java"),orig);
        var p=fixture.plan(root,ExtendedDeclarationsTest.xml("DATA"),fixture.record()); assertFalse(StarterJson.write(p.report()),p.conflicts()); Reconciler.apply(root,p,null);
        String once=Files.readString(d.resolve("Parent.java")), onceChild=Files.readString(d.resolve("Child.java")); var r=p.record();
        System.out.println("=== "+label+"\n"+once.replace("\r","\\r").replace("\t","\\t"));
        for(int i=0;i<3;i++) { var n=fixture.plan(root,ExtendedDeclarationsTest.xml("DATA"),r); assertFalse(n.conflicts()); assertTrue(label+" drift run "+i+": "+n.sources().keySet(),n.sources().isEmpty()); Reconciler.apply(root,n,rec(root)); r=n.record(); }
        // mode round trip on the existing class file too
        for(int i=0;i<3;i++) for(String mode:List.of("TRIGGER","DATA")) { var n=fixture.plan(root,ExtendedDeclarationsTest.xml(mode),r); assertFalse(StarterJson.write(n.report()),n.conflicts()); Reconciler.apply(root,n,rec(root)); r=n.record(); }
        assertEquals(label+" Parent drift after mode round trips",once,Files.readString(d.resolve("Parent.java")));
        assertEquals(label+" Child drift after mode round trips",onceChild,Files.readString(d.resolve("Child.java")));
        Path out=Files.createDirectories(root.resolve("classes")); compile(root,out,null);
    }
